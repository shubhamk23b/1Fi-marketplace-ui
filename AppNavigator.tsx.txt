import React from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { RootStackParamList } from './types';
import { MainTabNavigator } from './MainTabNavigator';
import { MarketplaceScreen } from '@/screens/Marketplace/MarketplaceScreen';
import { ProductDetailsScreen } from '@/screens/Marketplace/ProductDetailsScreen';
import { MarketplaceSuccessScreen } from '@/screens/Marketplace/MarketplaceSuccessScreen';

const Stack = createNativeStackNavigator<RootStackParamList>();

/**
 * Root navigator. The bottom tabs (existing app shell) live under
 * "MainTabs"; the Marketplace flow screens are pushed on top of it so
 * they get their own full-screen presentation with the bottom nav
 * hidden, matching standard fintech-commerce UX.
 */
export function AppNavigator() {
  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        <Stack.Screen name="MainTabs" component={MainTabNavigator} />
        <Stack.Screen name="Marketplace" component={MarketplaceScreen} />
        <Stack.Screen name="ProductDetails" component={ProductDetailsScreen} />
        <Stack.Screen
          name="MarketplaceSuccess"
          component={MarketplaceSuccessScreen}
          options={{ gestureEnabled: false }}
        />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
