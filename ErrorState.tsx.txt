import React from 'react';
import { StyleSheet, Text, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { colors } from '@/theme/colors';
import { spacing } from '@/theme/spacing';
import { MarketplaceButton } from './MarketplaceButton';

interface ErrorStateProps {
  title?: string;
  message?: string;
  onRetry: () => void;
}

/** Generic error state with a working retry CTA, used across the feature. */
export function ErrorState({
  title = 'Something went wrong',
  message = 'Unable to load marketplace products.',
  onRetry,
}: ErrorStateProps) {
  return (
    <View style={styles.container} accessibilityLabel={title}>
      <View style={styles.iconWrap}>
        <Ionicons name="alert-circle-outline" size={32} color={colors.error} />
      </View>
      <Text style={styles.title}>{title}</Text>
      <Text style={styles.message}>{message}</Text>
      <MarketplaceButton
        label="Try again"
        onPress={onRetry}
        variant="secondary"
        style={styles.button}
        accessibilityLabel="Retry loading marketplace"
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    alignItems: 'center',
    justifyContent: 'center',
    paddingHorizontal: spacing.xxxl,
    paddingVertical: spacing.xxxl * 1.5,
  },
  iconWrap: {
    width: 64,
    height: 64,
    borderRadius: 32,
    backgroundColor: colors.errorLight,
    alignItems: 'center',
    justifyContent: 'center',
    marginBottom: spacing.lg,
  },
  title: {
    fontSize: 16,
    fontWeight: '700',
    color: colors.darkText,
    marginBottom: spacing.xs,
  },
  message: {
    fontSize: 13,
    color: colors.secondaryText,
    textAlign: 'center',
    marginBottom: spacing.xl,
  },
  button: {
    minWidth: 160,
  },
});
